create view public.v_dashboard_alerts as
select
  'Missing Headcount'::text as alert,
  count(*) as count
from
  m_schools s
  left join d_school_headcount h on h.school_id = s.id
  and h.entry_date = (CURRENT_DATE + 1)
where
  s.status = true
  and h.id is null
union all
select
  'Menu Not Created'::text as alert,
  count(*) as count
from
  m_kitchen k
where
  not (
    exists (
      select
        1
      from
        d_menu m
      where
        m.kitchen_id = k.id
        and m.delivery_date = (CURRENT_DATE + 1)
    )
  );

 create view public.v_dashboard_daily_summary as
select
  k.id as kitchen_id,
  CURRENT_DATE as summary_date,
  count(distinct da.driver_id) as drivers_present,
  count(distinct rd.vehicle_id) as vehicles_used,
  COALESCE(sum(ft.amount), 0::numeric) as fuel_cost,
  COALESCE(sum(lc.quantity), 0::numeric) as lubricant_qty,
  count(distinct sfr.school_id) as schools_served
from
  m_kitchen k
  left join d_attendance da on da.kitchen_id = k.id
  and da.attendance_date = CURRENT_DATE
  left join d_route_data rd on rd.kitchen_id = k.id
  and rd.route_date = CURRENT_DATE
  left join d_fuel_tracking ft on ft.kitchen_id = k.id
  and ft.entry_date = CURRENT_DATE
  left join d_lubricants_consumption lc on lc.kitchen_id = k.id
  and lc.entry_date = CURRENT_DATE
  left join d_school_food_requirement sfr on sfr.delivery_date = CURRENT_DATE
group by
  k.id;

  create view public.v_dashboard_production as
select
  m.kitchen_id,
  m.delivery_date,
  d.id as dish_id,
  d.dish_name,
  d.unit,
  req.vessel_name,
  COALESCE(sum(req.quantity), 0::bigint) as total_vessels,
  COALESCE(
    sum(
      req.quantity::numeric * COALESCE(req.vessel_capacity, 1::numeric)
    ),
    0::numeric
  ) as total_quantity
from
  d_menu m
  join d_menu_dish_link mdl on mdl.menu_id = m.id
  join m_dish d on d.id = mdl.dish_id
  left join d_school_food_requirement req on req.kitchen_id = m.kitchen_id
  and req.delivery_date = m.delivery_date
  and req.dish_id = d.id
where
  m.is_holiday = false
group by
  m.kitchen_id,
  m.delivery_date,
  d.id,
  d.dish_name,
  d.unit,
  req.vessel_name;

  create view public.v_dashboard_summary as
select
  hc.entry_date as dashboard_date,
  ms.kitchen_id,
  sum(hc.total_count) as total_attendance,
  sum(ms.e_ps + ms.e_ups + ms.e_zphs) as total_enrollment,
  count(distinct hc.school_id) as running_schools,
  (
    select
      count(*) as count
    from
      m_schools s2
    where
      s2.kitchen_id = ms.kitchen_id
      and s2.status = true
  ) as total_schools,
  COALESCE(
    (
      select
        round(
          sum(sfr.vessel_capacity * sfr.quantity::numeric),
          2
        ) as round
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%rice%'::text
    ),
    0::numeric
  ) as rice_kg,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%rice%'::text
    ),
    0::bigint
  ) as rice_cauldron,
  COALESCE(
    (
      select
        round(
          sum(sfr.vessel_capacity * sfr.quantity::numeric),
          2
        ) as round
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and (
          d.dish_name::text ~~* '%dal%'::text
          or d.dish_name::text ~~* '%lentil%'::text
        )
    ),
    0::numeric
  ) as dal_kg,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and (
          d.dish_name::text ~~* '%dal%'::text
          or d.dish_name::text ~~* '%lentil%'::text
        )
    ),
    0::bigint
  ) as dal_cauldron,
  COALESCE(
    (
      select
        round(
          sum(sfr.vessel_capacity * sfr.quantity::numeric),
          2
        ) as round
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%curry%'::text
    ),
    0::numeric
  ) as curry_kg,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%curry%'::text
    ),
    0::bigint
  ) as curry_cauldron,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%snack%250%'::text
    ),
    0::bigint
  ) as snack_250,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%snack%500%'::text
    ),
    0::bigint
  ) as snack_500,
  COALESCE(
    (
      select
        sum(sfr.quantity) as sum
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%snack%1%kg%'::text
    ),
    0::bigint
  ) as snack_1kg,
  COALESCE(
    (
      select
        round(
          sum(sfr.vessel_capacity * sfr.quantity::numeric),
          2
        ) as round
      from
        d_school_food_requirement sfr
        join m_dish d on d.id = sfr.dish_id
      where
        sfr.kitchen_id = ms.kitchen_id
        and sfr.delivery_date = hc.entry_date
        and d.dish_name::text ~~* '%snack%'::text
    ),
    0::numeric
  ) as snack_kg,
  COALESCE(
    (
      select
        sum(rd.big_vessel_out) as sum
      from
        d_route_data rd
      where
        rd.kitchen_id = ms.kitchen_id
        and rd.route_date = hc.entry_date
    ),
    0::bigint
  ) as vessels_100pct,
  COALESCE(
    (
      select
        sum(rd.medium_vessel_out) as sum
      from
        d_route_data rd
      where
        rd.kitchen_id = ms.kitchen_id
        and rd.route_date = hc.entry_date
    ),
    0::bigint
  ) as vessels_75pct,
  COALESCE(
    (
      select
        sum(rd.small_vessel_out) as sum
      from
        d_route_data rd
      where
        rd.kitchen_id = ms.kitchen_id
        and rd.route_date = hc.entry_date
    ),
    0::bigint
  ) as vessels_50pct,
  COALESCE(
    (
      select
        sum(rd.mini_vessel_out) as sum
      from
        d_route_data rd
      where
        rd.kitchen_id = ms.kitchen_id
        and rd.route_date = hc.entry_date
    ),
    0::bigint
  ) as vessels_25pct,
  COALESCE(
    (
      select
        sum(
          rd.big_vessel_out + rd.medium_vessel_out + rd.small_vessel_out + rd.mini_vessel_out
        ) as sum
      from
        d_route_data rd
      where
        rd.kitchen_id = ms.kitchen_id
        and rd.route_date = hc.entry_date
    ),
    0::bigint
  ) as vessels_total
from
  d_school_headcount hc
  join m_schools ms on ms.id = hc.school_id
group by
  hc.entry_date,
  ms.kitchen_id;
